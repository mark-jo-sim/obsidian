Signature for `ConversationMessage`:

```python
@dataclass
class ConversationMessage:
    """A single message in a conversation with audio metadata."""
    sequence_position: int
    speaker_role: str  # "agent", "visitor", "system", "bot"
    text: str
    message_name: str = ""
    # Full resource name (e.g. customers/.../messages/{id})
    create_time: Optional[datetime.datetime] = None
    state: str = "COMPLETE"  # COMPLETE or INCOMPLETE
    audio_start_offset_ms: int = 0
    audio_end_offset_ms: int = 0
    transcription_words: list[ASRWord] = field(default_factory=list)

    @property
    def avg_word_confidence(self) -> Optional[float]:
        """Average confidence across all transcription words."""
        if not self.transcription_words:
            return None
        return sum(
	        w.confidence for w in self.transcription_words
        ) / len(self.transcription_words)

    @property
    def low_confidence_words(self) -> list[ASRWord]:
        """Words with confidence below 0.6."""
        return [w for w in self.transcription_words if w.confidence < 0.6]

    @property
    def has_transcription_data(self) -> bool:
        return len(self.transcription_words) > 0
```
