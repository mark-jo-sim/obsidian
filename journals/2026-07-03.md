#### Channel-wise audio validation

Deepgram API call:

```python
DG_URL = "https://api.deepgram.com/v1/listen"
DG_PARAMS_BASE = {
    "model": "nova-3-general",
    "language": "en",
    "smart_format": "true",
    "punctuate": "true",
    "numerals": "true",
    "filler_words": "true",
}

def transcribe_multichannel(webm_path: Path, api_key: str) -> tuple[str, str]:
    """
    POST full webm with multichannel=true;
    return (channel_0_text, channel_1_text).
    """
    headers = {"Authorization": f"Token {api_key}", "Content-Type": "audio/webm"}
    params = dict(DG_PARAMS_BASE)
    params["multichannel"] = "true"
    audio = webm_path.read_bytes()
    resp = httpx.post(
	    DG_URL,
	    params=params,
	    headers=headers,
	    content=audio,
	    timeout=300.0
	)
    resp.raise_for_status()
    data = resp.json()
    channels = data["results"]["channels"]
    if len(channels) < 2:
        raise RuntimeError(f"expected 2 channels, got {len(channels)}")
    c0 = channels[0]["alternatives"][0].get("transcript", "")
    c1 = channels[1]["alternatives"][0].get("transcript", "")
    return c0, c1
```

```python
DG_URL = "https://deepgram-all-nova-3-global-staging.voice-staging.internal.cresta.ai/v1/listen"
DG_PARAMS_BASE = {
    "model": "nova-3-general",

def transcribe_multichannel(
	webm_path: Path,
	dg_params: dict,
) -> tuple[str, str]:
    """
    POST full webm with multichannel=true;
    return (channel_0_text, channel_1_text).
	
	Arguments:
		webm_path: path to stereo webm file to transcribe
		dg_params: optional dict with following keys
			- model (default "nova-3-general")
		    - language  (default "en")
		    - smart_format (default "true")
		    - punctuate (default "true")
		    - numerals (default "true")
		    - filler_words (default "true")

    
    """
    headers = {"Content-Type": "audio/webm"}
    params = dict(DG_PARAMS_BASE)
    params.update(**dg_params)
    params["multichannel"] = "true"
    audio = webm_path.read_bytes()
    resp = httpx.post(
	    DG_URL,
	    params=params,
	    headers=headers,
	    content=audio,
	    timeout=300.0,
	)
    resp.raise_for_status()
    data = resp.json()
    channels = data["results"]["channels"]
    if len(channels) < 2:
        raise RuntimeError(f"expected 2 channels, got {len(channels)}")
    c0 = channels[0]["alternatives"][0].get("transcript", "")
    c1 = channels[1]["alternatives"][0].get("transcript", "")
    return c0, c1
```

Transcript data format:

```json
{
	"audio_uri": "...",
	"customer_id": "...",
	"conversation_id": "...",
	"messages": [
		{
			"speaker_role": "...",
			"text": "...",
			"timestamp": "..."
		},
		...
	],
	...
}

```

- Iterate through every line in the `downloaded_conversations.jsonl` file for each customer
- then get the text transcript for the respective speaker role from that JSONL object
- all we need to do is concatenate the text for every message object that belongs to the given role
- now we have a tuple of two strings
- we then compare that to the hypothesis string for each of the two channels returned by transcribe_multichannel
- then we compute the word error rate and infer the role.
